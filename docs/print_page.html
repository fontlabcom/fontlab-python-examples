    Advance input errors

    Narrow callback error handling in owned UI.

    The retained catalogue records these interaction checks: Apply bad, negative, huge and valid advances; errors stay in owned label, previous valid value survives; valid input clears error.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QWidget(), "Sample advance — callback errors") as handle:
        window = handle.widget
        editor = gui.QLineEdit(window)
        editor.setText("600")
        apply = gui.QPushButton("Apply sample advance (0–2000)", window)
        label = gui.QLabel("Advance: 600", window)
        layout = gui.QVBoxLayout(window)
        for control in (editor, apply, label):
            layout.addWidget(control)
        state = {"advance": 600, "error": ""}

        def update() -> None:
            try:
                advance = int(str(editor.text))
                if not 0 <= advance <= 2000:
                    msg = "Advance must be between 0 and 2000"
                    raise ValueError(msg)
            except ValueError:
                state["error"] = "Enter an integer from 0 to 2000"
                label.setText(state["error"])
                return
            state.update(advance=advance, error="")
            label.setText(f"Advance: {advance}")

        handle.connect_old(apply, "clicked()", update)
        handle.controls.update(input=editor, apply=apply)
        handle.inspect_ui = lambda: {**state, "text": str(label.text)}
        return handle

    Compiled wrapper and slot exposure

    C++ decorators, slots and metaobject exposure.

    The retained catalogue records these interaction checks: Click move: compiled non-slot move(QPoint) repositions preview; clear invokes native clear() slot; inspect its metaobject slot index; no custom C++ compilation.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned, required


def build(qt: Any) -> ExampleHandle:
    point_type = required(qt, "QtCore.QPoint")
    required(qt, "QtGui.QLineEdit")
    gui = qt.QtGui
    with owned(gui.QWidget(), "Compiled wrappers and native slots") as handle:
        window = handle.widget
        entry = gui.QLineEdit(window)
        entry.setText("A.alt")
        move = gui.QPushButton("Call wrapped move(QPoint)", window)
        clear = gui.QPushButton("Invoke native clear() slot", window)
        stage = gui.QWidget(window)
        stage.setMinimumHeight(80)
        preview = gui.QLabel("Synthetic glyph A", stage)
        status = gui.QLabel("Only existing compiled bindings are used", window)
        layout = gui.QVBoxLayout(window)
        for control in (entry, move, clear, stage, status):
            layout.addWidget(control)
        try:
            meta = entry.metaObject()
            slot_index = meta.indexOfSlot("clear()")
            if slot_index < 0 or not callable(preview.move):
                msg = "clear() or move(QPoint)"
                raise AttributeError(msg)
        except AttributeError as error:
            msg = (
                "Host must expose QLineEdit metaObject/clear() and QWidget.move(QPoint)"
            )
            raise RuntimeError(
                msg,
            ) from error
        state = {"moves": 0}

        def move_preview() -> None:
            state["moves"] += 1
            preview.move(point_type(20 * (state["moves"] % 3), 10))
            status.setText("Called a wrapped QWidget non-slot method")

        handle.connect_old(move, "clicked()", move_preview)
        handle.connect_old(clear, "clicked()", entry.clear)
        handle.controls.update(
            move=move,
            clear=clear,
            input=entry,
            preview=preview,
            status=status,
        )
        handle.inspect_ui = lambda: {
            **state,
            "input": str(entry.text),
            "native_slot": "clear()",
            "slot_index": int(slot_index),
            "position": [int(preview.pos.x()), int(preview.pos.y())],
            "native_class": str(meta.className()),
            "compiled_here": False,
        }
        return handle

    Contour canvas

    Custom painting with genuine QPainterPath.

    The retained catalogue records these interaction checks: Resize/repaint: closed curved outline remains visible; paint_count increases; close/relaunch without errors.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui

    class Canvas(gui.QWidget):
        """Draw an owned synthetic contour; FontLab shapes are never used."""

        def paintEvent(self, event: Any) -> None:
            gui.QWidget.paintEvent(self, event)
            painter = gui.QPainter(self)
            try:
                painter.setRenderHint(gui.QPainter.Antialiasing, True)
                painter.drawPath(self.path)
                self.paint_count += 1
            finally:
                painter.end()

    with owned(Canvas(), "Synthetic contour — QPainterPath") as handle:
        canvas = handle.widget
        canvas.paint_count = 0
        canvas.path = gui.QPainterPath()
        canvas.path.moveTo(60.0, 210.0)
        canvas.path.lineTo(60.0, 90.0)
        canvas.path.cubicTo(60.0, 25.0, 230.0, 25.0, 230.0, 90.0)
        canvas.path.lineTo(230.0, 210.0)
        canvas.path.closeSubpath()
        handle.release.append(lambda: setattr(canvas, "path", None))
        handle.inspect_ui = lambda: {
            "paint_count": canvas.paint_count,
            "path_elements": canvas.path.elementCount(),
            "synthetic": True,
        }
        return handle

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    names = ("A", "Aacute", "H", "O", "space")
    with owned(gui.QWidget(), "Sample glyph names — list model") as handle:

            "prefix": str(query.text),

    Loaded glyph metrics form

    QUiLoader and object-name lookup.

    The retained catalogue records these interaction checks: Set glyph_name.text and advance.value, click apply; result and inspection agree; load from another working directory; close/relaunch.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from pathlib import Path
from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned, required


def _load(qt: Any, parent: Any) -> Any:
    loader_type = required(qt, "QtUiTools.QUiLoader")
    path = Path(__file__).with_suffix(".ui")
    source, loader = qt.QtCore.QFile(str(path)), None
    try:
        loader = loader_type()
        if not source.open(qt.QtCore.QIODevice.ReadOnly):
            msg = f"Cannot open owned form {path.name}: {source.errorString()}"
            raise OSError(msg)
        form = loader.load(source, parent)
        if form is None:
            msg = f"Cannot load owned form: {loader.errorString()}"
            raise RuntimeError(msg)
        return form
    finally:
        source.close()
        source.deleteLater()
        if loader is not None:
            loader.deleteLater()


def build(qt: Any) -> ExampleHandle:
    with owned(qt.QtGui.QWidget(), "Loaded sample glyph form") as handle:
        form = _load(qt, handle.widget)
        qt.QtGui.QVBoxLayout(handle.widget).addWidget(form)
        names, pending = {}, [form]
        while pending:
            child = pending.pop()
            names[str(child.objectName)] = child
            pending.extend(child.children())
        needed = ("glyphName", "advanceBox", "applyButton", "resultLabel")
        if not all(name in names for name in needed):
            msg = "Owned form is missing required named controls"
            raise RuntimeError(msg)
        glyph, advance, apply, result = (names[name] for name in needed)
        state = {"glyph_name": "A", "advance": 600}

        def preview() -> None:
            state.update(glyph_name=str(glyph.text), advance=int(advance.value))
            result.setText(f"{state['glyph_name']}: advance {state['advance']}")

        handle.connect_old(apply, "clicked()", preview)
        handle.controls.update(
            glyph_name=glyph,
            advance=advance,
            apply=apply,
            result=result,
            form=form,
        )
        handle.inspect_ui = lambda: {**state, "result": str(result.text)}
        return handle

    launch("loaded-glyph-form")

    Synchronized advance controls

    Signal blocking to avoid control feedback.

    The retained catalogue records these interaction checks: Change spin and slider both directions; values agree, one update per change, no recursive callbacks.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QWidget(), "Sample advance — synchronized controls") as handle:
        window = handle.widget
        spin = gui.QSpinBox(window)
        slider = gui.QSlider(qt.QtCore.Qt.Horizontal, window)
        layout = gui.QVBoxLayout(window)
        for control in (spin, slider):
            control.setRange(0, 2000)
            control.setValue(600)
            layout.addWidget(control)
        state = {"updates": 0}

        def mirror(target: Any, value: int) -> None:
            previous = target.blockSignals(True)
            try:
                target.setValue(value)
                state["updates"] += 1
            finally:
                target.blockSignals(previous)

        handle.connect(spin.valueChanged, lambda value: mirror(slider, value))
        handle.connect(slider.valueChanged, lambda value: mirror(spin, value))
        handle.controls.update(spin=spin, slider=slider)
        handle.inspect_ui = lambda: {
            "spin": int(spin.value),
            "slider": int(slider.value),
            **state,
        }
        return handle

    Sample glyph metrics

    Read-only QAbstractTableModel callbacks.

    The retained catalogue records these interaction checks: Verify three rows, glyph/advance/LSB headers and numeric cells; no edits to user font.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui, core = qt.QtGui, qt.QtCore
    rows = (("A", 600, 40), ("H", 620, 50), ("O", 640, 30))
    headers = ("Glyph", "Advance", "LSB")

    class Metrics(gui.QAbstractTableModel):
        """Immutable sample rows; no access to FontLab font data."""

        def rowCount(self, parent: Any = None) -> int:
            return 0 if parent is not None and parent.isValid() else len(rows)

        def columnCount(self, parent: Any = None) -> int:
            return 0 if parent is not None and parent.isValid() else len(headers)

        def data(self, index: Any, role: int = core.Qt.DisplayRole) -> Any:
            if role != core.Qt.DisplayRole or not index.isValid():
                return None
            row, column = index.row(), index.column()
            if 0 <= row < len(rows) and 0 <= column < len(headers):
                return rows[row][column]
            return None

        def headerData(
            self,
            section: int,
            orientation: int,
            role: int = core.Qt.DisplayRole,
        ) -> Any:
            if (
                role == core.Qt.DisplayRole
                and orientation == core.Qt.Horizontal
                and 0 <= section < len(headers)
            ):
                return headers[section]
            return None

    with owned(gui.QTableView(), "Sample metrics — table model") as handle:
        model = Metrics(handle.widget)
        handle.widget.setModel(model)
        handle.model = model
        handle.inspect_ui = lambda: {
            "headers": list(headers),
            "rows": [list(row) for row in rows],
            "row_count": model.rowCount(),
            "column_count": model.columnCount(),
        }
        return handle

    Undo sample glyph metrics

    Undo integration on owned data.

    The retained catalogue records these interaction checks: Apply two advances, undo/redo; undo then apply a new value clears redo branch; native command overrides and label update work; close safely.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned, required


def build(qt: Any) -> ExampleHandle:
    stack_type = required(qt, "QtGui.QUndoStack")
    command_type = required(qt, "QtGui.QUndoCommand")
    gui = qt.QtGui
    state = {"advance": 600}

    class AdvanceCommand(command_type):
        """QUndoStack owns the command; callbacks modify only Python sample data."""

        def __init__(self, before: int, after: int) -> None:
            super().__init__(f"Advance {before} → {after}")
            self.before, self.after = before, after

        def undo(self) -> None:
            state["advance"] = self.before

        def redo(self) -> None:
            state["advance"] = self.after

    with owned(gui.QWidget(), "Undo sample glyph metrics") as handle:
        window = handle.widget
        advance = gui.QSpinBox(window)
        advance.setRange(0, 2000)
        advance.setValue(600)
        apply = gui.QPushButton("Apply sample advance", window)
        undo = gui.QPushButton("Undo", window)
        redo = gui.QPushButton("Redo", window)
        label = gui.QLabel("Advance: 600", window)
        layout = gui.QVBoxLayout(window)
        for control in (advance, apply, undo, redo, label):
            layout.addWidget(control)
        stack, commands = stack_type(window), []

        def update_label(*_args: Any) -> None:
            label.setText(f"Advance: {state['advance']}")

        def push() -> None:
            value = int(advance.value)
            if value == state["advance"]:
                return
            command = AdvanceCommand(state["advance"], value)
            commands.append(
                command,
            )  # Retain Python virtual receivers while C++ owns them.
            stack.push(command)

        def release() -> None:
            if not handle.owner.closed:
                stack.clear()  # All explicit connections have already been disconnected.
            commands.clear()

        handle.connect_old(apply, "clicked()", push)
        handle.connect_old(undo, "clicked()", stack.undo)
        handle.connect_old(redo, "clicked()", stack.redo)
        handle.connect(stack.indexChanged, update_label)
        handle.release.append(release)
        handle.controls.update(
            advance=advance,
            apply=apply,
            undo=undo,
            redo=redo,
            stack=stack,
            status=label,
        )
        handle.inspect_ui = lambda: {
            **state,
            "count": int(stack.count()),
            "index": int(
                stack.index(),
            ),  # a method in PythonQt; canUndo/canRedo are properties
            "can_undo": bool(stack.canUndo),
            "can_redo": bool(stack.canRedo),
            "text": str(label.text),
        }
        return handle

    launch("metrics-undo")

    Owned key observer

    QObject event filter scoped to owned widget.

    The retained catalogue records these interaction checks: Type in owned input: log updates without consuming keys; type elsewhere: log unchanged; remove filter on close.

    This is an archived example. Review its imports and host assumptions before running it.

    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui, core = qt.QtGui, qt.QtCore

    class KeyObserver(qt.private.QObject):
        """Observe keys only in one owned input, without consuming events."""

        def eventFilter(self, watched: Any, event: Any) -> bool:
            if watched is self.target and event.type() == core.QEvent.KeyPress:
                self.keys.append(str(event.text()))
                del self.keys[:-20]
                self.label.setText("Keys: " + " · ".join(self.keys))
            return False

        def release(self) -> None:
            self.target = self.label = None

    with owned(gui.QWidget(), "Owned glyph-name key observer") as handle:
        window = handle.widget
        editor = gui.QLineEdit(window)
        editor.setPlaceholderText("Type a glyph name here")
        label = gui.QLabel("Keys: ", window)
        layout = gui.QVBoxLayout(window)
        layout.addWidget(editor)
        layout.addWidget(label)
        observer = KeyObserver(window)
        observer.target, observer.label, observer.keys = editor, label, []
        handle.release.append(observer.release)
        handle.cleanup.append(
            lambda: (
                editor.removeEventFilter(observer) if not handle.owner.closed else None
            ),
        )
        editor.installEventFilter(observer)
        handle.controls.update(editor=editor, filter=observer, label=label)
        handle.inspect_ui = lambda: {
            "keys": list(observer.keys),
            "text": str(editor.text),
        }
        return handle

    The retained catalogue records these interaction checks: Run advance, assignment advance = 720, print(advance), and invalid syntax; all output stays in the owned console; reset restores the namespace.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QWidget(), "Owned sample-data console") as handle:
        window = handle.widget
        entry = gui.QLineEdit(window)
        entry.setText("advance")
        run = gui.QPushButton("Run short expression / statement", window)
        reset = gui.QPushButton("Reset owned namespace", window)
        output = gui.QPlainTextEdit(window)
        output.setReadOnly(True)
        output.setMaximumBlockCount(200)
        layout = gui.QVBoxLayout(window)
        for control in (entry, run, reset, output):
            layout.addWidget(control)
        namespace, state = {}, {"executions": 0, "error": ""}

        def print_owned(*values: Any, sep: str = " ", end: str = "\n") -> None:
            output.appendPlainText(sep.join(map(str, values)) + end.rstrip("\n"))

        def reset_namespace() -> None:
            namespace.clear()
            namespace.update(
                advance=600,
                glyph_names=["A", "H", "O"],
                __builtins__={
                    "len": len,
                    "sum": sum,
                    "min": min,
                    "max": max,
                    "round": round,
                    "str": str,
                    "int": int,
                    "float": float,
                    "sorted": sorted,
                    "print": print_owned,
                },
            )
            state.update(executions=0, error="")
            output.clear()

        def execute() -> None:
            source = str(entry.text)
            output.appendPlainText(">>> " + source)
            try:
                try:
                    compiled = compile(source, "<owned-console>", "eval")
                except SyntaxError:
                    exec(compile(source, "<owned-console>", "exec"), namespace)
                else:
                    result = eval(compiled, namespace)
                    if result is not None:
                        output.appendPlainText(repr(result))
            except Exception as error:
                state["error"] = f"{type(error).__name__}: {error}"
                output.appendPlainText(state["error"])
            else:
                state["error"] = ""
            state["executions"] += 1

        reset_namespace()
        handle.connect_old(run, "clicked()", execute)
        handle.connect(entry.returnPressed, execute)
        handle.connect_old(reset, "clicked()", reset_namespace)
        handle.release.append(namespace.clear)
        handle.controls.update(input=entry, run=run, reset=reset, output=output)
        handle.inspect_ui = lambda: {
            **state,
            "namespace_keys": sorted(k for k in namespace if k != "__builtins__"),
            "output": str(output.toPlainText()),
            "scope": "owned globals; no host imports",
        }
        return handle

    Owned menu plug-in tokens

    Registration and explicit unload.

    The retained catalogue records these interaction checks: Install twice: one menu/action token pair; trigger plugin_action; unload twice: zero tokens and detached menu; reinstall works; FontLab menus unchanged.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QMainWindow(), "Owned menu plug-in registry") as handle:
        window = handle.widget
        panel = gui.QWidget(window)
        window.setCentralWidget(panel)
        install_button = gui.QPushButton("Install sample plug-in", panel)
        unload_button = gui.QPushButton("Unload sample plug-in", panel)
        label = gui.QLabel("Unloaded", panel)
        layout = gui.QVBoxLayout(panel)
        for control in (install_button, unload_button, label):
            layout.addWidget(control)
        tokens, state, connection = {}, {"invocations": 0}, [None]

        def invoke(*_args: Any) -> None:
            state["invocations"] += 1
            label.setText(f"Sample action invoked {state['invocations']} times")

        def install() -> None:
            if tokens:
                return
            menu = window.menuBar().addMenu("Sample plug-in")
            tokens["proof-menu"] = menu
            action = gui.QAction("Preview sample advance", menu)
            tokens["advance-action"] = action
            signal = action.triggered
            handle.connect(signal, invoke)
            connection[0] = signal
            menu.addAction(action)
            handle.controls["plugin_action"] = action
            label.setText("Installed one menu/action token pair")

        def unload() -> None:
            if not tokens:
                return
            menu, action = tokens["proof-menu"], tokens.get("advance-action")
            if connection[0] is not None:
                handle.disconnect(connection[0], invoke)
                connection[0] = None
            if not handle.owner.closed:
                if action is not None:
                    menu.removeAction(action)
                window.menuBar().removeAction(menu.menuAction())
                label.setText("Unloaded")
            tokens.clear()
            handle.controls["plugin_action"] = None
            if not handle.owner.closed:
                menu.deleteLater()  # Menu owns its action; never delete the action twice.

        handle.cleanup.append(unload)
        handle.connect_old(install_button, "clicked()", install)
        handle.connect_old(unload_button, "clicked()", unload)
        handle.controls.update(
            install=install_button,
            unload=unload_button,
            plugin_action=None,
            status=label,
        )
        handle.inspect_ui = lambda: {
            **state,
            "tokens": sorted(tokens),
            "registered": bool(tokens),
        }
        return handle

    Worker glyph audit queue

    Worker-to-GUI queue hand-off.

    The retained catalogue records these interaction checks: Click start twice: one worker; wait for four GUI-delivered results; restart then stop or close immediately; worker joined, no Qt calls from worker.

    This is an archived example. Review its imports and host assumptions before running it.

    Source

    Python source
    from queue import Empty, Queue
from threading import Event, Thread
from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned

NAMES = ("A", "H", "O", "space")


def _produce(incoming: Queue, stop: Event) -> None:
    """Publish only plain Python data; no Qt objects enter this worker."""
    try:
        for name in NAMES:
            if stop.wait(0.03):
                return
            incoming.put_nowait(
                ("result", {"glyph": name, "name_score": sum(map(ord, name))}),
            )
        incoming.put_nowait(("done", None))
    except (TypeError, ValueError) as error:
        incoming.put_nowait(("error", str(error)))


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QWidget(), "Worker sample glyph audit") as handle:
        window, stop, incoming = handle.widget, Event(), Queue(maxsize=8)
        start_button = gui.QPushButton("Start sample audit", window)
        stop_button = gui.QPushButton("Stop audit", window)
        log = gui.QPlainTextEdit(window)
        log.setReadOnly(True)
        layout = gui.QVBoxLayout(window)
        for control in (start_button, stop_button, log):
            layout.addWidget(control)
        timer, worker = qt.QtCore.QTimer(window), None
        state = {"phase": "idle", "results": [], "error": ""}

        def stop_work() -> None:
            stop.set()
            if not handle.owner.closed:
                timer.stop()
            if worker is not None:
                worker.join(1.0)
                if worker.is_alive():
                    msg = "Sample audit worker did not stop; retaining owned UI"
                    raise RuntimeError(
                        msg,
                    )
            if state["phase"] == "running":
                state["phase"] = "stopped"

        def start() -> None:
            nonlocal worker
            if state["phase"] == "running":
                return
            stop_work()
            while not incoming.empty():
                incoming.get_nowait()
            stop.clear()
            state.update(phase="running", results=[], error="")
            log.clear()
            worker = Thread(
                target=_produce,
                args=(incoming, stop),
                name="insight-glyph-audit",
                daemon=True,
            )
            worker.start()
            timer.start(30)

        def deliver() -> None:
            for _ in range(8):
                try:
                    kind, value = incoming.get_nowait()
                except Empty:
                    return
                if kind == "result":
                    state["results"].append(value)
                    log.appendPlainText(f"{value['glyph']}: {value['name_score']}")
                else:
                    state.update(
                        phase="done" if kind == "done" else "error",
                        error=value or "",
                    )
                    stop_work()
                    return

        def release() -> None:
            nonlocal worker
            worker = None

        handle.cleanup.append(stop_work)
        handle.release.append(release)
        handle.connect_old(start_button, "clicked()", start)
        handle.connect_old(stop_button, "clicked()", stop_work)
        handle.connect(timer.timeout, deliver)
        handle.controls.update(
            start=start_button,
            stop=stop_button,
            timer=timer,
            log=log,
        )
        handle.inspect_ui = lambda: {
            **state,
            "results": [dict(row) for row in state["results"]],
            "worker_alive": bool(worker and worker.is_alive()),
            "timer_active": bool(timer.isActive()),
        }
        return handle

    Build and edit the documentation

    The source configuration is src_docs/mkdocs.yml. Authored pages are in src_docs/md. The build writes the static site to docs.

    ./build.sh
./build.sh --offline
./build.sh --serve

    The first build needs network access to install the packages pinned in uv.lock. Use --offline after those packages are cached. The build needs Python 3.12 or later and uv. It does not start FontLab or execute the scripts shown on a page.

    Serve starts a local preview. Live Edit is available only in that preview; it saves changes to the authored Markdown. The static site has no editing server. Edit generated catalogue pages through their source generator, because the next build replaces them.

    Include source and shared text

    Use codeinclude for an executable script. The plugin reads the actual source file, so the displayed code follows changes to that file. Do not paste a second copy of the script into Markdown. Use include-markdown for shared prose. Use mkdocstrings with Python inspection disabled when documenting an importable module. Menu scripts and archived host scripts are displayed as source instead.

    Name the action and its scope. Give one instruction per step where possible. Put prerequisites before the action. Keep exact API identifiers and interface labels. State a host limitation beside the operation that it affects. This is ASD-STE100-inspired writing; it is not a claim of controlled-language certification.

    Use page tags and categories to connect related tasks. The navigation follows .nav.yml. Use the print page for a combined printable copy.
