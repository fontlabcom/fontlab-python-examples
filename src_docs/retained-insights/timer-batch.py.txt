from typing import Any

from fontlab_python_examples.handle import ExampleHandle
from fontlab_python_examples.insights import owned


def build(qt: Any) -> ExampleHandle:
    gui = qt.QtGui
    with owned(gui.QWidget(), "Incremental sample audit — GUI timer") as handle:
        window = handle.widget
        label = gui.QLabel("Waiting for first tick", window)
        gui.QVBoxLayout(window).addWidget(label)
        timer = qt.QtCore.QTimer(window)
        names = ("A", "H", "O", "space")
        state = {"processed": 0, "glyph_names": []}

        def tick() -> None:
            index = state["processed"]
            if index >= len(names):
                timer.stop()
                return
            state["glyph_names"].append(names[index])
            state["processed"] += 1
            label.setText(f"Checked {names[index]} ({index + 1}/{len(names)})")
            if state["processed"] == len(names):
                timer.stop()

        handle.cleanup.append(lambda: timer.stop() if not handle.owner.closed else None)
        handle.connect(timer.timeout, tick)
        handle.controls.update(timer=timer, label=label)
        handle.inspect_ui = lambda: {
            "processed": state["processed"],
            "glyph_names": list(state["glyph_names"]),
            "timer_active": bool(timer.isActive()),
            "text": str(label.text),
        }
        timer.start(150)
        return handle


if __name__ == "__main__":
    from fontlab_python_examples import launch

    launch("timer-batch")
